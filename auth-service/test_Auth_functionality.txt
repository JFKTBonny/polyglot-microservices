# register
        curl -s -X POST http://localhost:3006/auth/register \
        -H "Content-Type: application/json" \
        -d '{
            "name": "Alice",
            "email": "alice@example.com",
            "password": "Secret123",
            "role": "user"
        }' | jq

# Login
        curl -s -X POST http://localhost:3006/auth/login \
        -H "Content-Type: application/json" \
        -d '{
            "email": "alice@example.com",
            "password": "Secret123"
        }' | jq
